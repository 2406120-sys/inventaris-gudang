-- Membuat Database CREATE DATABASE IF NOT EXISTS inventory_management; USE inventory_management; -- ==================== TABEL UTAMA ==================== -- 1. Tabel Kategori CREATE TABLE categories ( id INT PRIMARY KEY AUTO_INCREMENT, name VARCHAR(100) NOT NULL, description TEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP ); -- 2. Tabel Supplier CREATE TABLE suppliers ( id INT PRIMARY KEY AUTO_INCREMENT, name VARCHAR(150) NOT NULL, contact_person VARCHAR(100), phone VARCHAR(20), email VARCHAR(100), address TEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP ); -- 3. Tabel Produk CREATE TABLE products ( id INT PRIMARY KEY AUTO_INCREMENT, code VARCHAR(50) UNIQUE NOT NULL, name VARCHAR(200) NOT NULL, description TEXT, category_id INT, supplier_id INT, price DECIMAL(15,2) DEFAULT 0, stock INT DEFAULT 0, min_stock INT DEFAULT 5, location VARCHAR(50), created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE SET NULL, FOREIGN KEY (supplier_id) REFERENCES suppliers(id) ON DELETE SET NULL ); -- 4. Tabel Transaksi Masuk CREATE TABLE transactions_in ( id INT PRIMARY KEY AUTO_INCREMENT, product_id INT NOT NULL, quantity INT NOT NULL, price DECIMAL(15,2) NOT NULL, total DECIMAL(15,2) NOT NULL, transaction_date DATETIME DEFAULT CURRENT_TIMESTAMP, description VARCHAR(255), created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE ); -- 5. Tabel Transaksi Keluar CREATE TABLE transactions_out ( id INT PRIMARY KEY AUTO_INCREMENT, product_id INT NOT NULL, quantity INT NOT NULL, price DECIMAL(15,2) NOT NULL, total DECIMAL(15,2) NOT NULL, transaction_date DATETIME DEFAULT CURRENT_TIMESTAMP, description VARCHAR(255), created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE ); -- 6. Tabel User (untuk login) CREATE TABLE users ( id INT PRIMARY KEY AUTO_INCREMENT, username VARCHAR(50) UNIQUE NOT NULL, password VARCHAR(255) NOT NULL, full_name VARCHAR(150) NOT NULL, email VARCHAR(100) UNIQUE, role ENUM('admin','staff','manager') DEFAULT 'staff', is_active BOOLEAN DEFAULT TRUE, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP ); -- 7. Tabel Log Aktivitas CREATE TABLE activity_logs ( id INT PRIMARY KEY AUTO_INCREMENT, user_id INT, action VARCHAR(50) NOT NULL, table_name VARCHAR(50), record_id INT, old_value TEXT, new_value TEXT, ip_address VARCHAR(45), created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE SET NULL ); -- ==================== TRIGGER ==================== -- Trigger untuk update stok produk saat transaksi masuk DELIMITER $$ CREATE TRIGGER update_stock_in AFTER INSERT ON transactions_in FOR EACH ROW BEGIN UPDATE products SET stock = stock + NEW.quantity WHERE id = NEW.product_id; END$$ DELIMITER ; -- Trigger untuk update stok produk saat transaksi keluar DELIMITER $$ CREATE TRIGGER update_stock_out AFTER INSERT ON transactions_out FOR EACH ROW BEGIN UPDATE products SET stock = stock - NEW.quantity WHERE id = NEW.product_id; END$$ DELIMITER ; -- ==================== VIEW ==================== -- View untuk melihat stok produk dengan informasi kategori dan supplier CREATE VIEW view_product_stock AS SELECT p.id, p.code, p.name, p.stock, p.min_stock, p.price, c.name AS category_name, s.name AS supplier_name, p.location, CASE WHEN p.stock <= p.min_stock THEN 'Low Stock' WHEN p.stock <= p.min_stock * 3 THEN 'Medium Stock' ELSE 'High Stock' END AS stock_status FROM products p LEFT JOIN categories c ON p.category_id = c.id LEFT JOIN suppliers s ON p.supplier_id = s.id; -- View untuk laporan transaksi masuk CREATE VIEW view_transactions_in AS SELECT ti.id, p.code AS product_code, p.name AS product_name, ti.quantity, ti.price, ti.total, ti.transaction_date, ti.description FROM transactions_in ti JOIN products p ON ti.product_id = p.id ORDER BY ti.transaction_date DESC; -- View untuk laporan transaksi keluar CREATE VIEW view_transactions_out AS SELECT to2.id, p.code AS product_code, p.name AS product_name, to2.quantity, to2.price, to2.total, to2.transaction_date, to2.description FROM transactions_out to2 JOIN products p ON to2.product_id = p.id ORDER BY to2.transaction_date DESC; -- ==================== INDEX ==================== CREATE INDEX idx_product_name ON products(name); CREATE INDEX idx_product_code ON products(code); CREATE INDEX idx_transaction_date_in ON transactions_in(transaction_date);
